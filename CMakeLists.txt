cmake_minimum_required(VERSION 3.24)
project(CodexSwitcher VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(CODEX_SWITCHER_TESTS "Build synthetic test entry points (Windows)" OFF)
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/imgui")
set(IMGUI_SOURCES ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/imgui_demo.cpp)
set(APP_SOURCES src/studio.cpp src/settings.cpp src/accounts.cpp src/relative_time.cpp)
if(WIN32)
    list(APPEND IMGUI_SOURCES ${IMGUI_DIR}/backends/imgui_impl_win32.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp)
    list(APPEND APP_SOURCES src/main.cpp src/codex.cpp src/switch_process.cpp src/platform_windows.cpp)
else()
    find_package(OpenGL REQUIRED)
    find_package(glfw3 3.3 REQUIRED)
    find_package(OpenSSL REQUIRED)
    find_package(Threads REQUIRED)
    list(APPEND IMGUI_SOURCES ${IMGUI_DIR}/backends/imgui_impl_glfw.cpp ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp)
    list(APPEND APP_SOURCES src/main_glfw.cpp src/codex_posix.cpp src/switch_posix.cpp src/platform_posix.cpp)
endif()
add_library(imgui STATIC ${IMGUI_SOURCES})
target_include_directories(imgui PUBLIC ${IMGUI_DIR} ${IMGUI_DIR}/backends)
if(WIN32)
    add_executable(CodexSwitcher WIN32 ${APP_SOURCES})
else()
    add_executable(CodexSwitcher ${APP_SOURCES})
    target_link_libraries(imgui PUBLIC glfw OpenGL::GL ${CMAKE_DL_LIBS})
endif()
target_include_directories(CodexSwitcher PRIVATE vendor)
target_link_libraries(CodexSwitcher PRIVATE imgui)
if(MSVC)
    target_compile_options(CodexSwitcher PRIVATE /W4 /utf-8)
    target_compile_definitions(CodexSwitcher PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(CodexSwitcher PRIVATE d3d11 dxgi d3dcompiler dwmapi windowscodecs ole32 shell32 advapi32 crypt32 bcrypt wintrust version)
else()
    target_compile_options(CodexSwitcher PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
    target_link_libraries(CodexSwitcher PRIVATE OpenSSL::Crypto Threads::Threads)
    if(APPLE)
        target_link_libraries(CodexSwitcher PRIVATE "-framework Security" "-framework CoreFoundation")
        set_target_properties(CodexSwitcher PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_BUNDLE_NAME "Codex Switcher" MACOSX_BUNDLE_GUI_IDENTIFIER "org.codex-switcher.app" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" OUTPUT_NAME "CodexSwitcher")
    else()
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(SECRET REQUIRED IMPORTED_TARGET libsecret-1)
        target_link_libraries(CodexSwitcher PRIVATE PkgConfig::SECRET)
        set_target_properties(CodexSwitcher PROPERTIES OUTPUT_NAME "codex-switcher")
    endif()
endif()
set_target_properties(CodexSwitcher PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/bin")
if(APPLE)
    add_custom_command(TARGET CodexSwitcher POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_BUNDLE_DIR:CodexSwitcher>/Contents/Resources/assets")
    install(TARGETS CodexSwitcher BUNDLE DESTINATION .)
else()
    add_custom_command(TARGET CodexSwitcher POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:CodexSwitcher>/assets")
    install(TARGETS CodexSwitcher RUNTIME DESTINATION .)
    install(DIRECTORY assets DESTINATION .)
endif()
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md SECURITY.md DESTINATION .)
install(FILES vendor/imgui/LICENSE.txt DESTINATION licenses RENAME Dear-ImGui-LICENSE.txt)
install(FILES vendor/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json-LICENSE.txt)
install(FILES licenses/original-switcher-LICENSE.txt DESTINATION licenses)
if(CODEX_SWITCHER_TESTS AND WIN32)
    enable_testing()
    add_test(NAME LocalStorageAndSettings COMMAND CodexSwitcher --self-test)
endif()
set(CPACK_PACKAGE_NAME "codex-switcher")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_GENERATOR "ZIP")
include(CPack)
